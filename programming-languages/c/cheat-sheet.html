The following program allocates an array of structs, sorts it with qsort and a comparison callback, prints the result and frees the memory (compile with clang -std=c23 -Wall -Wextra):

#include <stdio.h>
#include <stdlib.h>

typedef struct Item {
    const char *name;
    int qty;
} Item;

static int by_qty_desc(const void *a, const void *b)
{
    const Item *x = a;
    const Item *y = b;
    return (y->qty > x->qty) - (y->qty < x->qty);
}

int main(void)
{
    const size_t count = 3;
    Item *items = malloc(count * sizeof *items);
    if (items == nullptr) {
        return EXIT_FAILURE;
    }

    items[0] = (Item){ "apple", 3 };
    items[1] = (Item){ "plum", 12 };
    items[2] = (Item){ "pear", 7 };

    qsort(items, count, sizeof *items, by_qty_desc);

    for (size_t i = 0; i < count; i++) {
        printf("%-6s %2d\n", items[i].name, items[i].qty);
    }

    free(items);
    return EXIT_SUCCESS;
}

It prints:

plum   12
pear    7
apple   3
