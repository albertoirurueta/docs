The Mental Models Behind the Sheet

A self-check you can run

The script below turns twelve of the sheet’s claims into assertions. If one fails on your interpreter, the corresponding page explains the behaviour.

"""Executable versions of the cheat sheet's key claims. Each check prints OK or raises AssertionError."""
import functools

def check(label, condition):
    assert condition, label
    print("OK  ", label)

a = [1]; b = a; c = list(a)
check("assignment aliases, copy() does not", b is a and c is not a)
check("tuples hash, lists do not", hash((1, 2)) is not None and getattr([], "__hash__") is None)
check("and/or return an operand", ("" or "x") == "x" and (1 and 2) == 2)
check("default args are evaluated once", (lambda f: f() is f())(lambda x=[]: x))
check("closures capture variables", [f() for f in [lambda: i for i in range(2)]] == [1, 1])
check("generators are single-pass", (lambda g: (list(g), list(g)))(x for x in "ab") == (["a", "b"], []))
check("sort is stable", sorted([("b", 1), ("a", 1)], key=lambda t: t[1]) == [("b", 1), ("a", 1)])

class A:
    def who(self): return "A"
class B(A):
    def who(self): return "B" + super().who()
check("super follows the MRO", B().who() == "BA" and [k.__name__ for k in B.__mro__] == ["B", "A", "object"])

class P:
    @property
    def x(self): return 1
check("property is a data descriptor on the class", hasattr(type(P.x), "__set__"))

@functools.lru_cache(maxsize=None)
def sq(n): return n * n
sq(3); sq(3)
check("lru_cache caches by arguments", sq.cache_info().hits == 1)

with open(__file__) as f:
    pass
check("with closes the file", f.closed)

try:
    {}["k"]
except KeyError as e:
    check("exceptions are objects with a type", type(e).__name__ == "KeyError" and e.args == ("k",))

OK   closures capture variables
