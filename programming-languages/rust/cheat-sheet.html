Worked example

The example parses two numbers with ?, handles the error case in match, and chains iterator adapters.

use std::num::ParseIntError;

fn total_pages(a: &str, b: &str) -> Result<u32, ParseIntError> {
    let x: u32 = a.trim().parse()?;
    let y: u32 = b.trim().parse()?;
    Ok(x + y)
}

fn main() {
    println!("{:?}", total_pages("120", " 80 "));
    match total_pages("120", "abc") {
        Ok(n) => println!("{n} pages"),
        Err(e) => println!("error: {e}"),
    }

    let titles = vec!["Dune", "Emma", "Ulysses"];
    let long: Vec<usize> = titles.iter().map(|t| t.len()).filter(|&n| n > 4).collect();
    println!("{long:?}");
}
Ok(200)
error: invalid digit found in string
[7]

Each ? returns the Err early, converting it with From; the iterator chain is lazy and runs only when collect() drives it.
