The following short module combines destructuring, a class with a private field and a getter, template literals and `await` (top-level `await` requires an ES module):

[source,javascript]
----
const inventory = [
  { name: "apple", qty: 3, price: 0.5 },
  { name: "pear", qty: 0, price: 0.75 },
  { name: "plum", qty: 12, price: 0.2 },
];

class Cart {
  #items = [];
  add({ name, qty, price }) {
    if (qty > 0) this.#items.push({ name, total: qty * price });
    return this;
  }
  get total() {
    return this.#items.reduce((sum, { total }) => sum + total, 0);
  }
}

const cart = new Cart();
inventory.forEach((item) => cart.add(item));
console.log(`Total: ${cart.total.toFixed(2)}`);

const delay = (ms, value) => new Promise((resolve) => setTimeout(resolve, ms, value));
const result = await delay(10, "done");
console.log(result);
----

Running it prints:

[source,text]
----
Total: 3.90
done
----
