Worked example

A complete, runnable check of the routing and extractor rules above: a router with one typed path capture is called in-process with tower::ServiceExt::oneshot, so no port is opened. It needs axum, tokio (with the macros and rt-multi-thread features), serde_json, tower (with util) and http-body-util.

use axum::{
    body::Body,
    extract::Path,
    http::Request,
    routing::get,
    Json, Router,
};
use http_body_util::BodyExt;
use serde_json::{json, Value};
use tower::ServiceExt;

async fn book(Path(id): Path<u32>) -> Json<Value> {
    Json(json!({ "id": id, "title": "Rust in Action" }))
}

#[tokio::main]
async fn main() {
    let app = Router::new().route("/books/{id}", get(book));

    for uri in ["/books/42", "/books/abc", "/missing"] {
        let request = Request::get(uri).body(Body::empty()).unwrap();
        let response = app.clone().oneshot(request).await.unwrap();
        let status = response.status();
        if status.is_success() {
            let body = response.into_body().collect().await.unwrap().to_bytes();
            println!("GET {uri} -> {status} {}", String::from_utf8_lossy(&body));
        } else {
            println!("GET {uri} -> {status}");
        }
    }
}
GET /books/42 -> 200 OK {"id":42,"title":"Rust in Action"}
GET /books/abc -> 400 Bad Request
GET /missing -> 404 Not Found

The typed Path<u32> rejects abc with 400 before the handler runs, and an unmatched path falls through to the default 404 fallback.
