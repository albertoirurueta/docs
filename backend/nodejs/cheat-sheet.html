Worked example

One small ES module that exercises four of the essentials above: concurrent promises, error codes from a failed file read, a stream pipeline and a one-shot event listener. Save it as cheat.mjs and run it with node cheat.mjs.

cheat.mjs
import { EventEmitter } from 'node:events';
import { readFile } from 'node:fs/promises';
import { setTimeout as sleep } from 'node:timers/promises';
import { pipeline } from 'node:stream/promises';
import { Readable, Transform } from 'node:stream';

// Promises: two 100 ms waits run concurrently with Promise.all
const started = performance.now();
const results = await Promise.all([sleep(100, 'a'), sleep(100, 'b')]);
const elapsed = performance.now() - started;
console.log('results:', results, 'concurrent:', elapsed < 190);

// Errors: a failed read rejects with a code
try {
  await readFile(new URL('./missing.txt', import.meta.url));
} catch (err) {
  console.log('caught:', err.code);
}

// Streams: pipeline handles backpressure and cleanup
const upper = new Transform({
  transform(chunk, _enc, cb) { cb(null, chunk.toString().toUpperCase()); },
});
let out = '';
await pipeline(
  Readable.from(['node', ' ', 'streams']),
  upper,
  async function (source) { for await (const c of source) out += c; },
);

// Events: a one-shot listener
const bus = new EventEmitter();
bus.once('done', (text) => console.log('event done:', text));
bus.emit('done', out);
console.log('listeners left:', bus.listenerCount('done'));

Output on Node.js 24 (the program uses only APIs that exist on 22 and 26 as well):

results: [ 'a', 'b' ] concurrent: true
caught: ENOENT
event done: NODE STREAMS
listeners left: 0

The two 100 ms waits finish together because Promise.all() starts both before awaiting either, the missing file surfaces as a rejection carrying err.code, pipeline() closes every stream in the chain when it ends, and once() removes its listener after the first call.
