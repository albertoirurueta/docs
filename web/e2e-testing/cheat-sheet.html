Worked example

The same check written for the three tools against the Bookshelf application: open the catalogue, search for a title and assert on the single result.

// Playwright: tests/search.spec.ts
import { test, expect } from '@playwright/test';

test('search finds a book', async ({ page }) => {
  await page.goto('/books');
  await page.getByTestId('search-input').fill('quiet');
  await page.getByRole('button', { name: 'Search' }).click();
  await expect(page.getByTestId('book-title')).toHaveText('The Quiet Compiler');
});
// Cypress: cypress/e2e/search.cy.ts
describe('catalogue', () => {
  it('search finds a book', () => {
    cy.visit('/books');
    cy.get('[data-testid=search-input]').type('quiet');
    cy.contains('button', 'Search').click();
    cy.get('[data-testid=book-title]').should('have.text', 'The Quiet Compiler');
  });
});
# Selenium with pytest: tests/test_search.py
from selenium.webdriver.common.by import By
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.support.ui import WebDriverWait

def test_search_finds_a_book(driver, base_url):
    driver.get(f"{base_url}/books")
    driver.find_element(By.CSS_SELECTOR, "[data-testid=search-input]").send_keys("quiet")
    driver.find_element(By.XPATH, "//button[normalize-space()='Search']").click()
    title = WebDriverWait(driver, 10).until(
        EC.visibility_of_element_located((By.CSS_SELECTOR, "[data-testid=book-title]"))
    )
    assert title.text == "The Quiet Compiler"

Playwright and Cypress retry the assertion on their own; Selenium needs the explicit wait. The driver and base_url fixtures belong to the project’s conftest.py.
